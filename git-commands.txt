git config --global --list 	--> verificar nuestras configuraciones
git config --list 		--> verificar todas las configuraciones

git init   		--> inicializar un repositorio local
git status  		--> ver el estado actual, ve si hay cambios
git diff 		--> muestra la diferencia o cambios de los archivos
git log 		--> ver el historial
git log --oneline 	--> muestra el resumen de forma compacta
git show HEAD~1:nombreArchivo.txt  --> muestra lo que estaba en "1" commit antes al actual
git show HEAD:nombreArchivo.txt		--> ver ultimo archivo
git show f898e57:git-commands.txt	--> ver archivo de un commit especifico
git diff HEAD~1 HEAD 	--> muestra la diferencia entre las ultimas 2 versiones


git rm --cached <NombreArchivo>	    --> Para guardar un archivo que esta en .gitignore
git commit -m "Stop tracking file"  --> se guarda el archivo con formato dentro del .gitignore
************************
git add . 			--> agrega y prepara archivos todo o por nombre de archivo
git commit -m "descripcion" 	--> guarda
************************
git branch 		  	--> lista las ramas en el proyecto
git branch feature-1 	  	--> crea la rama
git checkout feature-1    	--> cambia a la rama feature-1 también $ git switch feature-1
git switch -c <rama>		--> cambia y crea la rama
git merge feature-aerodas 	--> desde el master se hace el merge de la rama feature-aerodas
git branch -d feature-aerodas 	--> borrar una rama "feature-aerodas"
